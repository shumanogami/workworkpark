# ボット計測（調整用の控え）

調整ログ.md の数字を測ったコード。プロジェクトの外（スクラッチパッド）に置いて、次のように回す。

```sh
WANDER=0.5 ROT=1 SK=good,good,mid,bad N=600 npx vitest run --config vitest.config.mjs --root <フォルダ>
```

設定ファイル `vitest.config.mjs` は `export default { test: { globals: true, include: ["sim.test.ts"], testTimeout: 900000 } }`。結果は `result.txt` に出る。

type Skill = { name: string; interval: number; noise: number; mistake: number; jumpReach: number; maxRise: number }
const SKILLS: Record<string, Skill> = {
  good: { name: 'good', interval: 0.05, noise: 4, mistake: 0.005, jumpReach: 90, maxRise: 100 },
  mid: { name: 'mid', interval: 0.15, noise: 14, mistake: 0.04, jumpReach: 75, maxRise: 90 },
  bad: { name: 'bad', interval: 0.3, noise: 30, mistake: 0.12, jumpReach: 60, maxRise: 80 },

function rng(seed: number) { let s = seed | 0; return () => { s = (s + 0x6d2b79f5) | 0; let t = Math.imul(s ^ (s >>> 15), s | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61); return ((t ^ (t >>> 14)) >>> 0) / 4294967296 } }

function makeBot(skill: Skill, seed: number) {
  const rand = rng(seed)
  let next = 0
  let cmd = { left: false, right: false, jump: false }
  let wantJump = false
  let lockId: string | null = null
  let wasStanding = true
  return (s: MagmaState, id: string) => {
    const r = s.runners.find((x) => x.playerId === id)!
    const t = s.scroll.elapsed
    const justLanded = r.standingOn !== null && !wasStanding
    wasStanding = r.standingOn !== null
    if (t >= next || justLanded) {
      next = t + skill.interval
      const feet = r.y + r.height
      const cx = r.x + r.width / 2
      const locked = r.standingOn === null && lockId ? s.platforms.find((p) => p.id === lockId) : undefined
      const cands = locked ? [locked] : s.platforms.filter((p) => p.id !== r.standingOn && p.y < feet - 8 && p.y > feet - skill.maxRise)
      let tgt = cands.sort((a, b) => a.y - b.y || Math.abs(a.x + a.width / 2 - cx) - Math.abs(b.x + b.width / 2 - cx))[0]
      // 高いところを狙うが、横に遠すぎるものは避ける
      const near = cands.filter((p) => Math.max(p.x - cx, cx - (p.x + p.width), 0) < 140)
      if (near.length > 1 && rand() < Number(process.env.WANDER ?? 0)) tgt = near[Math.floor(rand() * near.length)]
      else if (near.length) tgt = near.sort((a, b) => a.y - b.y || Math.abs(a.x + a.width / 2 - cx) - Math.abs(b.x + b.width / 2 - cx))[0]
      if (tgt && !locked) lockId = tgt.id
      if (process.env.STUCK && cands.length >= 0 && id === 'g' && t > 18 && t < 19.6) (globalThis as any).DBG = ((globalThis as any).DBG ?? '') + `t=${t.toFixed(2)} stand=${r.standingOn} feet=${feet.toFixed(0)} cx=${cx.toFixed(0)} tgt=${tgt?.id}(${tgt?.x},${tgt?.y},w${tgt?.width}) locked=${!!locked} ncand=${cands.length}\n`
      if (!tgt) { cmd = { left: false, right: false, jump: false }; wantJump = false; return cmd }
      const tx = tgt.x + tgt.width / 2 + (rand() - 0.5) * 2 * skill.noise
      const dx = tx - cx
      const edge = Math.max(tgt.x - cx, cx - (tgt.x + tgt.width), 0)
      cmd = { left: dx < -10, right: dx > 10, jump: false }
      wantJump = edge <= skill.jumpReach
      if (rand() < skill.mistake) wantJump = !wantJump
    }
    return { ...cmd, jump: wantJump && r.standingOn !== null && !r.jumpHeld }
  }
}

type Res = { dur: number; winners: string[]; elim: Record<string, number> }
function play(ids: string[], skills: string[], seed: number): Res {
  let s = createMagmaState(ids, seed)
  const bots = ids.map((id, i) => makeBot(SKILLS[skills[i]], seed * 100 + i))
  while (!s.finished && s.scroll.elapsed < 120) {
    const inputs: Record<string, any> = {}
    ids.forEach((id, i) => { inputs[id] = bots[i](s, id) })
    s = step(s, inputs, FIXED_DT)
  }
  const elim: Record<string, number> = {}
  for (const e of s.survival.eliminated) elim[e.playerId] = e.at
  return { dur: s.scroll.elapsed, winners: s.winners, elim }
}

it('sim', () => {
  if (process.env.STUCK) return
  const N = Number(process.env.N ?? 150)
  const ids = ['g1', 'g2', 'm1', 'b1']
  const sk = (process.env.SK ?? 'good,good,mid,bad').split(',')
  const wins: Record<string, number> = {}; const early: Record<string, number> = {}; const sum: Record<string, number> = {}
  const skillWin: Record<string, number> = {}, skillN: Record<string, number> = {}, skillSum: Record<string, number> = {}, skillEarly: Record<string, number> = {}
  let draws = 0; const durs: number[] = []; let unfinished = 0
  for (let seed = 1; seed <= N; seed++) {
    const rot = process.env.ROT ? seed % 4 : 0
    const sks = sk.map((_, i) => sk[(i + rot) % 4])
    const r = play(ids, sks, seed)
    ids.forEach((id, i) => { skillWin[sks[i]] = (skillWin[sks[i]] ?? 0) + (r.winners.includes(id) ? 1 / r.winners.length : 0); skillN[sks[i]] = (skillN[sks[i]] ?? 0) + 1; skillSum[sks[i]] = (skillSum[sks[i]] ?? 0) + (r.elim[id] ?? r.dur); if ((r.elim[id] ?? r.dur) < 10) skillEarly[sks[i]] = (skillEarly[sks[i]] ?? 0) + 1 })
    durs.push(r.dur)

    for (const w of r.winners) wins[w] = (wins[w] ?? 0) + 1 / r.winners.length
    ids.forEach((id) => { const at = r.elim[id] ?? r.dur; sum[id] = (sum[id] ?? 0) + at; if (at < 10) early[id] = (early[id] ?? 0) + 1 })
  }
  durs.sort((a, b) => a - b)
  const q = (p: number) => durs[Math.floor(p * (N - 1))].toFixed(1)
  const f = (o: Record<string, number>, k: string, d = 1) => ((o[k] ?? 0) / d).toFixed(2)

  for (const id of ids) lines.push(`${id}(${sk[ids.indexOf(id)]}) 勝率${(((wins[id] ?? 0) / N) * 100).toFixed(1)}% 平均生存${f(sum, id, N)}s 10秒以内脱落${(((early[id] ?? 0) / N) * 100).toFixed(0)}%`)

})
